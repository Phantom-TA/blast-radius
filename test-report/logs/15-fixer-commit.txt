    Send an idempotency key on every charge attempt to stop double charges
    
    A provider timeout does not mean the charge failed: the money can move and
    only the response is lost. charge_order retried without an idempotency key,
    so each retry after a timeout created a second charge (e.g. ORD-100953).
    
    charge_order now derives one key per order and sends it on every attempt;
    the provider already answers a repeat key with the original charge.
    
    Adds a regression test (timeout then retry -> exactly one charge) and a
    test that every attempt sends the same key.
    
    Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>

+"""Regression test: a provider timeout followed by a retry must not charge twice (ORD-100953)."""
+
+import pytest
+
+from ..charge_client import charge_order
+from ..provider import FakeProvider
+from ..store import connect, create_db
+
+
+@pytest.fixture
+def conn(tmp_path):
+    db_path = str(tmp_path / "payments.db")
+    create_db(db_path)
+    conn = connect(db_path)
+    conn.execute("INSERT INTO customers VALUES ('CUS-1', 'Ana Silva', 'ana@example.com')")
+    conn.execute(
+        "INSERT INTO orders VALUES ('ORD-1', 'CUS-1', 'Desk lamp', 4900, '2026-10-01T10:00:00+00:00')"
+    )
+    yield conn
+    conn.close()
+
+
+def test_timeout_then_retry_charges_the_order_once(conn):
+    calls = []
+
+    def time_out_first_attempt(order_id):
+        # The charge goes through on the first attempt, but its response is lost.
+        calls.append(order_id)
+        return len(calls) == 1
+
+    provider = FakeProvider(conn, should_timeout=time_out_first_attempt)
+
+    charge_id = charge_order(provider, "ORD-1", 4900)
+
+    charges = conn.execute("SELECT id FROM charges WHERE order_id = 'ORD-1'").fetchall()
+    assert calls, "the first attempt should have timed out"
+    assert [row["id"] for row in charges] == [charge_id]
