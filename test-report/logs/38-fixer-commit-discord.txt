    Fix double charge when a provider timeout is retried
    
    charge_order retried after ProviderTimeout without an idempotency key, but
    a timeout can lose the response after the charge already went through, so
    the retry created a second charge for the same order (e.g. ORD-100953).
    
    Every attempt to charge an order now sends the same idempotency key
    ("charge:<order_id>"), so the provider returns the original charge on a
    repeat instead of charging again.
    
    Adds test_timeout_then_retry_charges_once as a regression test.

+"""Regression test: a timeout followed by a retry must not charge the order twice."""
+
+import pytest
+
+from ..charge_client import charge_order
+from ..provider import FakeProvider
+from ..store import connect, create_db
+
+
+@pytest.fixture
+def conn(tmp_path):
+    db_path = str(tmp_path / "payments.db")
+    create_db(db_path)
+    conn = connect(db_path)
+    conn.execute("INSERT INTO customers VALUES ('CUS-1', 'Ana Silva', 'ana@example.com')")
+    conn.execute(
+        "INSERT INTO orders VALUES ('ORD-1', 'CUS-1', 'Desk lamp', 4900, '2026-10-01T10:00:00+00:00')"
+    )
+    yield conn
+    conn.close()
+
+
+def test_timeout_then_retry_charges_once(conn):
+    calls = []
+
+    def first_call_times_out(order_id):
+        # The provider records the charge, then the response is lost once.
+        calls.append(order_id)
+        return len(calls) == 1
+
+    provider = FakeProvider(conn, should_timeout=first_call_times_out)
+
+    charge_id = charge_order(provider, "ORD-1", 4900)
+
+    rows = conn.execute("SELECT id, amount_cents FROM charges WHERE order_id = 'ORD-1'").fetchall()
+    assert len(rows) == 1, f"expected exactly one charge, found {len(rows)}"
+    assert rows[0]["id"] == charge_id
+    assert rows[0]["amount_cents"] == 4900
